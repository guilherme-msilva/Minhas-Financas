
CREATE TABLE `categorias` (
  `id` int(11) NOT NULL AUTO_INCREMENT,
  `nome` varchar(255) DEFAULT NULL,
  `id_pai` int(11) DEFAULT NULL,
  `cor` char(9) DEFAULT NULL,
  `id_user` int(11) DEFAULT NULL,
  `icone` varchar(255) DEFAULT NULL,
  PRIMARY KEY (`id`)
) ENGINE=InnoDB AUTO_INCREMENT=198 DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_general_ci;

CREATE TABLE `contas` (
  `id` int(11) NOT NULL AUTO_INCREMENT,
  `nome` varchar(100) DEFAULT NULL,
  `saldo_inicial` decimal(10,2) DEFAULT 0.00,
  `cor` char(9) DEFAULT NULL,
  `img` varchar(250) DEFAULT NULL,
  `status` tinyint(1) DEFAULT 1,
  `id_user` int(11) DEFAULT NULL,
  PRIMARY KEY (`id`)
) ENGINE=InnoDB AUTO_INCREMENT=20 DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_general_ci;

CREATE TABLE `transacoes` (
  `id` int(11) NOT NULL AUTO_INCREMENT,
  `data` date NOT NULL,
  `valor` decimal(15,2) DEFAULT NULL,
  `descricao` varchar(255) DEFAULT NULL,
  `idcategoria` int(11) DEFAULT NULL,
  `idconta` int(11) DEFAULT NULL,
  `iduser` int(11) DEFAULT NULL,
  `consolidada` tinyint(1) DEFAULT NULL,
  `recorrencias` int(11) DEFAULT NULL,
  `idpai` int(11) DEFAULT NULL,
  `notas` varchar(255) DEFAULT NULL,
  `id_grupo_recorrencia` varchar(255) DEFAULT NULL, 
  `parcela_recorrencia` int(11) DEFAULT NULL,
  `parcela_fim` int(11) DEFAULT NULL,
  PRIMARY KEY (`id`)
) ENGINE=InnoDB AUTO_INCREMENT=26822 DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_general_ci;

CREATE TABLE `usuarios` (
  `id` int(11) NOT NULL AUTO_INCREMENT,
  `nome` varchar(255) DEFAULT NULL,
  `email` varchar(255) NOT NULL,
  `senha` varchar(255) DEFAULT NULL,
  `tema` varchar(100) DEFAULT NULL,
  `google_sheets_id` varchar(255) DEFAULT NULL,
  PRIMARY KEY (`id`),
  UNIQUE KEY `usuarios_unique` (`email`)
) ENGINE=InnoDB AUTO_INCREMENT=2 DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_general_ci;

CREATE TABLE `categorias_investimento` (

  `id_pai` int(11) DEFAULT NULL,
  PRIMARY KEY (`id`)
